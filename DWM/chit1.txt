

CREATE TABLE DimTime (
    TimeKey INT PRIMARY KEY,
    FullDate DATE NOT NULL,
    DayNumber INT,
    MonthName VARCHAR(20),
    QuarterNumber INT,
    CalendarYear INT
);

CREATE TABLE DimHotel (
    HotelKey INT PRIMARY KEY,
    HotelName VARCHAR(100) NOT NULL,
    City VARCHAR(50),
    StateName VARCHAR(50)
);

CREATE TABLE DimRoom (
    RoomKey INT PRIMARY KEY,
    RoomNumber VARCHAR(10),
    RoomType VARCHAR(30),
    FloorNo INT,
    Wing VARCHAR(30)
);

CREATE TABLE DimGuest (
    GuestKey INT PRIMARY KEY,
    GuestName VARCHAR(100),
    GuestType VARCHAR(30)
);

CREATE TABLE DimBookingChannel (
    ChannelKey INT PRIMARY KEY,
    ChannelName VARCHAR(50),
    ChannelCategory VARCHAR(50)
);

CREATE TABLE FactHotelOccupancy (
    FactKey INT PRIMARY KEY,
    TimeKey INT NOT NULL,
    HotelKey INT NOT NULL,
    RoomKey INT NOT NULL,
    GuestKey INT NOT NULL,
    ChannelKey INT NOT NULL,
    RoomsOccupied INT NOT NULL,
    AvailableRooms INT NOT NULL,
    NightsStayed INT NOT NULL,
    RoomRevenue DECIMAL(12,2) NOT NULL,
    OccupancyPercent DECIMAL(5,2) NOT NULL,

    FOREIGN KEY (TimeKey) REFERENCES DimTime(TimeKey),
    FOREIGN KEY (HotelKey) REFERENCES DimHotel(HotelKey),
    FOREIGN KEY (RoomKey) REFERENCES DimRoom(RoomKey),
    FOREIGN KEY (GuestKey) REFERENCES DimGuest(GuestKey),
    FOREIGN KEY (ChannelKey) REFERENCES DimBookingChannel(ChannelKey)
);

CREATE TABLE SnowTime (

    FullDate DATE NOT NULL,
    DayNumber INT,

    QuarterNumber INT,
    CalendarYear INT
);

CREATE TABLE SnowState (
    StateKey INT PRIMARY KEY,
    StateName VARCHAR(50) NOT NULL
);

CREATE TABLE SnowCity (
    CityKey INT PRIMARY KEY,
    CityName VARCHAR(50) NOT NULL,
    StateKey INT NOT NULL,
    FOREIGN KEY (StateKey) REFERENCES SnowState(StateKey)
);

CREATE TABLE SnowHotel (
    HotelKey INT PRIMARY KEY,
    HotelName VARCHAR(100) NOT NULL,
    CityKey INT NOT NULL,
    FOREIGN KEY (CityKey) REFERENCES SnowCity(CityKey)
);

CREATE TABLE SnowRoomType (
    RoomTypeKey INT PRIMARY KEY,
    RoomTypeName VARCHAR(30) NOT NULL
);

CREATE TABLE SnowFloorWing (
    FloorWingKey INT PRIMARY KEY,
    FloorNo INT,
    Wing VARCHAR(30)
);

CREATE TABLE SnowRoom (
    RoomKey INT PRIMARY KEY,
    RoomNumber VARCHAR(10),
    RoomTypeKey INT NOT NULL,
    FloorWingKey INT NOT NULL,
    FOREIGN KEY (RoomTypeKey) REFERENCES SnowRoomType(RoomTypeKey),
    FOREIGN KEY (FloorWingKey) REFERENCES SnowFloorWing(FloorWingKey)
);

CREATE TABLE SnowGuest (
    GuestKey INT PRIMARY KEY,
    GuestName VARCHAR(100),
    GuestType VARCHAR(30)
);

CREATE TABLE SnowBookingChannel (
    ChannelKey INT PRIMARY KEY,
    ChannelName VARCHAR(50),
    ChannelCategory VARCHAR(50)
);

CREATE TABLE SnowFactHotelOccupancy (
    FactKey INT PRIMARY KEY,
    TimeKey INT NOT NULL,
    HotelKey INT NOT NULL,
    RoomKey INT NOT NULL,
    GuestKey INT NOT NULL,
    ChannelKey INT NOT NULL,
    RoomsOccupied INT NOT NULL,
    AvailableRooms INT NOT NULL,
    NightsStayed INT NOT NULL,
    RoomRevenue DECIMAL(12,2) NOT NULL,
    OccupancyPercent DECIMAL(5,2) NOT NULL,

    FOREIGN KEY (TimeKey) REFERENCES SnowTime(TimeKey),
    FOREIGN KEY (HotelKey) REFERENCES SnowHotel(HotelKey),
    FOREIGN KEY (RoomKey) REFERENCES SnowRoom(RoomKey),
    FOREIGN KEY (GuestKey) REFERENCES SnowGuest(GuestKey),
    FOREIGN KEY (ChannelKey) REFERENCES SnowBookingChannel(ChannelKey)
);

SELECT
    T.FullDate,
    T.MonthName,
    T.CalendarYear,
    H.HotelName,
    H.City,
    H.StateName,
    R.RoomNumber,
    R.RoomType,
    R.FloorNo,
    R.Wing,
    G.GuestName,
    G.GuestType,
    B.ChannelName,
    B.ChannelCategory,
    F.RoomsOccupied,
    F.AvailableRooms,
    F.NightsStayed,
    F.RoomRevenue,
    F.OccupancyPercent
FROM FactHotelOccupancy F
JOIN DimTime T
    ON F.TimeKey = T.TimeKey
JOIN DimHotel H
    ON F.HotelKey = H.HotelKey
JOIN DimRoom R
    ON F.RoomKey = R.RoomKey
JOIN DimGuest G
    ON F.GuestKey = G.GuestKey
JOIN DimBookingChannel B
    ON F.ChannelKey = B.ChannelKey;

    C.CityName,

    FW.FloorNo,
    FW.Wing,

    G.GuestType,
    B.ChannelName,
    B.ChannelCategory,
    F.RoomsOccupied,
    F.AvailableRooms,
    F.NightsStayed,
    F.RoomRevenue,
    F.OccupancyPercent

    ON R.FloorWingKey = FW.FloorWingKey
JOIN SnowGuest G
    ON F.GuestKey = G.GuestKey
JOIN SnowBookingChannel B
    ON F.ChannelKey = B.ChannelKey;




    






























# Star Schema: Hotel Occupancy

One central fact table (`FactHotelOccupancy`) connected directly to 5 dimension tables.

Legend: `PK` = Primary Key, `FK` = Foreign Key, `▲ ▼` = arrow points to the dimension table (the PK side).

```
┌────────────────────────┐   ┌────────────────────────┐   ┌────────────────────────┐
│ DimTime                │   │ DimHotel               │   │ DimRoom                │
├────────────────────────┤   ├────────────────────────┤   ├────────────────────────┤
│ PK TimeKey             │   │ PK HotelKey            │   │ PK RoomKey             │
│    FullDate            │   │    HotelName           │   │    RoomNumber          │
│    DayNumber           │   │    City                │   │    RoomType            │
│    MonthName           │   │    StateName           │   │    FloorNo             │
│    QuarterNumber       │   │                        │   │    Wing                │
│    CalendarYear        │   │                        │   │                        │
└───────────┬────────────┘   └───────────┬────────────┘   └───────────┬────────────┘
            ▲                            ▲                            ▲
            │                            │                            │
            │ TimeKey                    │ HotelKey                   │ RoomKey
            │                            │                            │
      ┌─────┴────────────────────────────┴────────────────────────────┴──────┐
      │ FactHotelOccupancy  (FACT TABLE)                                     │
      ├──────────────────────────────────────────────────────────────────────┤
      │ Keys                               Measures                          │
      │ PK FactKey                           RoomsOccupied                   │
      │ FK TimeKey                           AvailableRooms                  │
      │ FK HotelKey                          NightsStayed                    │
      │ FK RoomKey                           RoomRevenue                     │
      │ FK GuestKey                          OccupancyPercent                │
      │ FK ChannelKey                                                        │
      └───────────────────┬──────────────────────────────┬───────────────────┘
                          │                              │
                          │ GuestKey                     │ ChannelKey
                          ▼                              ▼
              ┌────────────────────────┐     ┌────────────────────────┐
              │ DimGuest               │     │ DimBookingChannel      │
              ├────────────────────────┤     ├────────────────────────┤
              │ PK GuestKey            │     │ PK ChannelKey          │
              │    GuestName           │     │    ChannelName         │
              │    GuestType           │     │    ChannelCategory     │
              └────────────────────────┘     └────────────────────────┘

```

## Connections

| Fact Table FK                     | Dimension Table PK              |
|-----------------------------------|---------------------------------|
| FactHotelOccupancy.TimeKey        | DimTime.TimeKey                 |
| FactHotelOccupancy.HotelKey       | DimHotel.HotelKey               |
| FactHotelOccupancy.RoomKey        | DimRoom.RoomKey                 |
| FactHotelOccupancy.GuestKey       | DimGuest.GuestKey               |
| FactHotelOccupancy.ChannelKey     | DimBookingChannel.ChannelKey    |

Total: 6 tables (1 fact + 5 dimensions), all dimensions are denormalized (no dimension links to another dimension).





















# Snowflake Schema: Hotel Occupancy

Same fact table, but `Hotel` and `Room` dimensions are normalized into sub-dimensions:

- Hotel → City → State
- Room → RoomType and Room → FloorWing

Legend: `PK` = Primary Key, `FK` = Foreign Key, `▲ ▼ ►` = arrow points to the parent table (the PK side).

```
┌──────────────────────┐     ┌──────────────────────┐     ┌──────────────────────┐     ┌──────────────────────┐

├──────────────────────┤     ├──────────────────────┤     ├──────────────────────┤     ├──────────────────────┤
│ PK TimeKey           │     │ PK HotelKey          │     │ PK CityKey           │     │ PK StateKey          │
│    FullDate          │     │    HotelName         │     │    CityName          │     │    StateName         │

│    MonthName         │     └──────────┬───────────┘     └──────────────────────┘

└──────────┬───────────┘                │
           ▲                            │

           │                            │
```
