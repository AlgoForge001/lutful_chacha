/* =====================================================
   PART A: CREATE STAR SCHEMA
   ===================================================== */

DROP TABLE IF EXISTS Star_FactAdmissions;
DROP TABLE IF EXISTS Star_DimDiagnosis;
DROP TABLE IF EXISTS Star_DimDepartment;
DROP TABLE IF EXISTS Star_DimDoctor;
DROP TABLE IF EXISTS Star_DimPatient;
DROP TABLE IF EXISTS Star_DimTime;


/* 1. TIME DIMENSION */

CREATE TABLE Star_DimTime (
    TimeKey INTEGER PRIMARY KEY,
    FullDate DATE NOT NULL,
    DayNumber INTEGER,
    MonthName VARCHAR(20),
    QuarterNumber INTEGER,
    CalendarYear INTEGER
);


/* 2. PATIENT DIMENSION */

CREATE TABLE Star_DimPatient (
    PatientKey INTEGER PRIMARY KEY,
    PatientName VARCHAR(100),
    Gender VARCHAR(20),
    Age INTEGER,
    Area VARCHAR(100),
    City VARCHAR(50)
);


/* 3. DOCTOR DIMENSION */

CREATE TABLE Star_DimDoctor (
    DoctorKey INTEGER PRIMARY KEY,
    DoctorName VARCHAR(100),
    Qualification VARCHAR(50),
    Specialisation VARCHAR(100),
    DepartmentName VARCHAR(100)
);


/* 4. DEPARTMENT DIMENSION */

CREATE TABLE Star_DimDepartment (
    DepartmentKey INTEGER PRIMARY KEY,
    DepartmentName VARCHAR(100),
    DepartmentLocation VARCHAR(100)
);


/* 5. DIAGNOSIS DIMENSION */

CREATE TABLE Star_DimDiagnosis (
    DiagnosisKey INTEGER PRIMARY KEY,
    DiagnosisName VARCHAR(100),
    DiagnosisCategory VARCHAR(100),
    Severity VARCHAR(30)
);


/* 6. FACT TABLE */

CREATE TABLE Star_FactAdmissions (
    AdmissionKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    PatientKey INTEGER NOT NULL,
    DoctorKey INTEGER NOT NULL,
    DepartmentKey INTEGER NOT NULL,
    DiagnosisKey INTEGER NOT NULL,

    AdmissionsCount INTEGER NOT NULL,
    LengthOfStayDays INTEGER NOT NULL,
    TreatmentCost DECIMAL(12,2) NOT NULL,
    BillAmount DECIMAL(12,2) NOT NULL,

    FOREIGN KEY (TimeKey)
        REFERENCES Star_DimTime(TimeKey),

    FOREIGN KEY (PatientKey)
        REFERENCES Star_DimPatient(PatientKey),

    FOREIGN KEY (DoctorKey)
        REFERENCES Star_DimDoctor(DoctorKey),

    FOREIGN KEY (DepartmentKey)
        REFERENCES Star_DimDepartment(DepartmentKey),

    FOREIGN KEY (DiagnosisKey)
        REFERENCES Star_DimDiagnosis(DiagnosisKey)
);

/* =====================================================
   PART D: STAR SCHEMA JOIN
   FACT + ALL DIMENSIONS
   ===================================================== */

SELECT
    T.FullDate,
    T.MonthName,
    T.CalendarYear,

    P.PatientName,
    P.Gender,
    P.Age,
    P.Area,
    P.City,

    DOC.DoctorName,
    DOC.Qualification,
    DOC.Specialisation,

    DEP.DepartmentName,
    DEP.DepartmentLocation,

    DIA.DiagnosisName,
    DIA.DiagnosisCategory,
    DIA.Severity,

    F.AdmissionsCount,
    F.LengthOfStayDays,
    F.TreatmentCost,
    F.BillAmount

FROM Star_FactAdmissions F

JOIN Star_DimTime T
    ON F.TimeKey = T.TimeKey

JOIN Star_DimPatient P
    ON F.PatientKey = P.PatientKey

JOIN Star_DimDoctor DOC
    ON F.DoctorKey = DOC.DoctorKey

JOIN Star_DimDepartment DEP
    ON F.DepartmentKey = DEP.DepartmentKey

JOIN Star_DimDiagnosis DIA
    ON F.DiagnosisKey = DIA.DiagnosisKey;


/* =====================================================
   PART E: CREATE SNOWFLAKE SCHEMA
   ===================================================== */

DROP TABLE IF EXISTS Snow_FactAdmissions;
DROP TABLE IF EXISTS Snow_Diagnosis;
DROP TABLE IF EXISTS Snow_Doctor;
DROP TABLE IF EXISTS Snow_Specialisation;
DROP TABLE IF EXISTS Snow_Department;
DROP TABLE IF EXISTS Snow_Patient;
DROP TABLE IF EXISTS Snow_Area;
DROP TABLE IF EXISTS Snow_City;
DROP TABLE IF EXISTS Snow_Time;


/* 1. TIME DIMENSION */

CREATE TABLE Snow_Time (
    TimeKey INTEGER PRIMARY KEY,
    FullDate DATE NOT NULL,
    DayNumber INTEGER,
    MonthName VARCHAR(20),
    QuarterNumber INTEGER,
    CalendarYear INTEGER
);


/* 2. CITY DIMENSION */

CREATE TABLE Snow_City (
    CityKey INTEGER PRIMARY KEY,
    CityName VARCHAR(50) NOT NULL
);


/* 3. AREA DIMENSION */

CREATE TABLE Snow_Area (
    AreaKey INTEGER PRIMARY KEY,
    AreaName VARCHAR(100) NOT NULL,
    CityKey INTEGER NOT NULL,

    FOREIGN KEY (CityKey)
        REFERENCES Snow_City(CityKey)
);


/* 4. PATIENT DIMENSION */

CREATE TABLE Snow_Patient (
    PatientKey INTEGER PRIMARY KEY,
    PatientName VARCHAR(100),
    Gender VARCHAR(20),
    Age INTEGER,
    AreaKey INTEGER NOT NULL,

    FOREIGN KEY (AreaKey)
        REFERENCES Snow_Area(AreaKey)
);


/* 5. DEPARTMENT DIMENSION */

CREATE TABLE Snow_Department (
    DepartmentKey INTEGER PRIMARY KEY,
    DepartmentName VARCHAR(100),
    DepartmentLocation VARCHAR(100)
);


/* 6. SPECIALISATION DIMENSION */

CREATE TABLE Snow_Specialisation (
    SpecialisationKey INTEGER PRIMARY KEY,
    SpecialisationName VARCHAR(100)
);


/* 7. DOCTOR DIMENSION */

CREATE TABLE Snow_Doctor (
    DoctorKey INTEGER PRIMARY KEY,
    DoctorName VARCHAR(100),
    Qualification VARCHAR(50),
    SpecialisationKey INTEGER NOT NULL,
    DepartmentKey INTEGER NOT NULL,

    FOREIGN KEY (SpecialisationKey)
        REFERENCES Snow_Specialisation(SpecialisationKey),

    FOREIGN KEY (DepartmentKey)
        REFERENCES Snow_Department(DepartmentKey)
);


/* 8. DIAGNOSIS DIMENSION */

CREATE TABLE Snow_Diagnosis (
    DiagnosisKey INTEGER PRIMARY KEY,
    DiagnosisName VARCHAR(100),
    DiagnosisCategory VARCHAR(100),
    Severity VARCHAR(30)
);


/* 9. SNOWFLAKE FACT TABLE */

CREATE TABLE Snow_FactAdmissions (
    AdmissionKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    PatientKey INTEGER NOT NULL,
    DoctorKey INTEGER NOT NULL,
    DepartmentKey INTEGER NOT NULL,
    DiagnosisKey INTEGER NOT NULL,

    AdmissionsCount INTEGER NOT NULL,
    LengthOfStayDays INTEGER NOT NULL,
    TreatmentCost DECIMAL(12,2) NOT NULL,
    BillAmount DECIMAL(12,2) NOT NULL,

    FOREIGN KEY (TimeKey)
        REFERENCES Snow_Time(TimeKey),

    FOREIGN KEY (PatientKey)
        REFERENCES Snow_Patient(PatientKey),

    FOREIGN KEY (DoctorKey)
        REFERENCES Snow_Doctor(DoctorKey),

    FOREIGN KEY (DepartmentKey)
        REFERENCES Snow_Department(DepartmentKey),

    FOREIGN KEY (DiagnosisKey)
        REFERENCES Snow_Diagnosis(DiagnosisKey)
);

/* =====================================================
   PART H: SNOWFLAKE SCHEMA JOIN
   FACT + ALL DIMENSIONS
   ===================================================== */

SELECT
    T.FullDate,
    T.MonthName,
    T.CalendarYear,

    P.PatientName,
    P.Gender,
    P.Age,
    A.AreaName,
    C.CityName,

    DOC.DoctorName,
    DOC.Qualification,
    SPEC.SpecialisationName,

    DEP.DepartmentName,
    DEP.DepartmentLocation,

    DIA.DiagnosisName,
    DIA.DiagnosisCategory,
    DIA.Severity,

    F.AdmissionsCount,
    F.LengthOfStayDays,
    F.TreatmentCost,
    F.BillAmount

FROM Snow_FactAdmissions F

JOIN Snow_Time T
    ON F.TimeKey = T.TimeKey

JOIN Snow_Patient P
    ON F.PatientKey = P.PatientKey

JOIN Snow_Area A
    ON P.AreaKey = A.AreaKey

JOIN Snow_City C
    ON A.CityKey = C.CityKey

JOIN Snow_Doctor DOC
    ON F.DoctorKey = DOC.DoctorKey

JOIN Snow_Specialisation SPEC
    ON DOC.SpecialisationKey = SPEC.SpecialisationKey

JOIN Snow_Department DEP
    ON F.DepartmentKey = DEP.DepartmentKey

JOIN Snow_Diagnosis DIA
    ON F.DiagnosisKey = DIA.DiagnosisKey;
