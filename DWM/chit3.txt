/* Remove old tables so the script can be rerun */

DROP TABLE IF EXISTS Star_FactSales;
DROP TABLE IF EXISTS Star_DimPromotion;
DROP TABLE IF EXISTS Star_DimCustomer;
DROP TABLE IF EXISTS Star_DimStore;
DROP TABLE IF EXISTS Star_DimProduct;
DROP TABLE IF EXISTS Star_DimTime;


/* 1. TIME DIMENSION */

CREATE TABLE Star_DimTime (
    TimeKey INTEGER PRIMARY KEY,
    FullDate DATE NOT NULL,
    DayNumber INTEGER,
    MonthName VARCHAR(20),
    QuarterNumber INTEGER,
    CalendarYear INTEGER
);


/* 2. PRODUCT DIMENSION */

CREATE TABLE Star_DimProduct (
    ProductKey INTEGER PRIMARY KEY,
    ProductName VARCHAR(100),
    CategoryName VARCHAR(50),
    BrandName VARCHAR(50)
);


/* 3. STORE DIMENSION */

CREATE TABLE Star_DimStore (
    StoreKey INTEGER PRIMARY KEY,
    StoreName VARCHAR(100),
    City VARCHAR(50),
    StateName VARCHAR(50)
);


/* 4. CUSTOMER DIMENSION */

CREATE TABLE Star_DimCustomer (
    CustomerKey INTEGER PRIMARY KEY,
    CustomerName VARCHAR(100),
    CustomerType VARCHAR(50),
    Email VARCHAR(100)
);


/* 5. PROMOTION DIMENSION */

CREATE TABLE Star_DimPromotion (
    PromotionKey INTEGER PRIMARY KEY,
    PromotionName VARCHAR(100),
    PromotionType VARCHAR(50),
    StartDate DATE,
    EndDate DATE
);


/* 6. STAR FACT TABLE */

CREATE TABLE Star_FactSales (
    SalesKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    ProductKey INTEGER NOT NULL,
    StoreKey INTEGER NOT NULL,
    CustomerKey INTEGER NOT NULL,
    PromotionKey INTEGER NOT NULL,

    QuantitySold INTEGER NOT NULL,
    SalesAmount DECIMAL(12,2) NOT NULL,
    DiscountAmount DECIMAL(12,2) NOT NULL,
    Profit DECIMAL(12,2) NOT NULL,

    FOREIGN KEY (TimeKey)
        REFERENCES Star_DimTime(TimeKey),

    FOREIGN KEY (ProductKey)
        REFERENCES Star_DimProduct(ProductKey),

    FOREIGN KEY (StoreKey)
        REFERENCES Star_DimStore(StoreKey),

    FOREIGN KEY (CustomerKey)
        REFERENCES Star_DimCustomer(CustomerKey),

    FOREIGN KEY (PromotionKey)
        REFERENCES Star_DimPromotion(PromotionKey)
);

/* =====================================================
   PART D: STAR SCHEMA JOIN QUERY
   ===================================================== */

SELECT
    T.FullDate,
    T.MonthName,
    T.CalendarYear,

    P.ProductName,
    P.CategoryName,
    P.BrandName,

    S.StoreName,
    S.City,
    S.StateName,

    C.CustomerName,
    C.CustomerType,

    PR.PromotionName,
    PR.PromotionType,

    F.QuantitySold,
    F.SalesAmount,
    F.DiscountAmount,
    F.Profit

FROM Star_FactSales F

JOIN Star_DimTime T
    ON F.TimeKey = T.TimeKey

JOIN Star_DimProduct P
    ON F.ProductKey = P.ProductKey

JOIN Star_DimStore S
    ON F.StoreKey = S.StoreKey

JOIN Star_DimCustomer C
    ON F.CustomerKey = C.CustomerKey

JOIN Star_DimPromotion PR
    ON F.PromotionKey = PR.PromotionKey;

    TimeKey INTEGER PRIMARY KEY,
    FullDate DATE NOT NULL,
    DayNumber INTEGER,
    MonthName VARCHAR(20),
    QuarterNumber INTEGER,
    CalendarYear INTEGER
);


/* 2. PRODUCT CATEGORY */

CREATE TABLE Snow_Category (
    CategoryKey INTEGER PRIMARY KEY,
    CategoryName VARCHAR(50) NOT NULL
);


/* 3. PRODUCT BRAND */

CREATE TABLE Snow_Brand (
    BrandKey INTEGER PRIMARY KEY,
    BrandName VARCHAR(50) NOT NULL
);


/* 4. PRODUCT DIMENSION */

CREATE TABLE Snow_Product (
    ProductKey INTEGER PRIMARY KEY,
    ProductName VARCHAR(100) NOT NULL,
    CategoryKey INTEGER NOT NULL,
    BrandKey INTEGER NOT NULL,

    FOREIGN KEY (CategoryKey)
        REFERENCES Snow_Category(CategoryKey),

    FOREIGN KEY (BrandKey)
        REFERENCES Snow_Brand(BrandKey)

CREATE TABLE Snow_State (
    StateKey INTEGER PRIMARY KEY,
    StateName VARCHAR(50) NOT NULL
);


/* 6. CITY DIMENSION */

CREATE TABLE Snow_City (
    CityKey INTEGER PRIMARY KEY,
    CityName VARCHAR(50) NOT NULL,
    StateKey INTEGER NOT NULL,

    FOREIGN KEY (StateKey)
        REFERENCES Snow_State(StateKey)
);


/* 7. STORE DIMENSION */

CREATE TABLE Snow_Store (
    StoreKey INTEGER PRIMARY KEY,
    StoreName VARCHAR(100) NOT NULL,
    CityKey INTEGER NOT NULL,

    FOREIGN KEY (CityKey)
        REFERENCES Snow_City(CityKey)
);


/* 8. CUSTOMER DIMENSION */

CREATE TABLE Snow_Customer (
    CustomerKey INTEGER PRIMARY KEY,
    CustomerName VARCHAR(100),
    CustomerType VARCHAR(50),
    Email VARCHAR(100)
);


/* 9. PROMOTION DIMENSION */

CREATE TABLE Snow_Promotion (
    PromotionKey INTEGER PRIMARY KEY,
    PromotionName VARCHAR(100),
    PromotionType VARCHAR(50),
    StartDate DATE,
    EndDate DATE
);


/* 10. SNOWFLAKE FACT TABLE */

CREATE TABLE Snow_FactSales (
    SalesKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    ProductKey INTEGER NOT NULL,
    StoreKey INTEGER NOT NULL,
    CustomerKey INTEGER NOT NULL,
    PromotionKey INTEGER NOT NULL,

    QuantitySold INTEGER NOT NULL,
    SalesAmount DECIMAL(12,2) NOT NULL,
    DiscountAmount DECIMAL(12,2) NOT NULL,
    Profit DECIMAL(12,2) NOT NULL,

    FOREIGN KEY (TimeKey)
        REFERENCES Snow_Time(TimeKey),

    FOREIGN KEY (ProductKey)
        REFERENCES Snow_Product(ProductKey),

    FOREIGN KEY (StoreKey)
        REFERENCES Snow_Store(StoreKey),

    FOREIGN KEY (CustomerKey)
        REFERENCES Snow_Customer(CustomerKey),

    FOREIGN KEY (PromotionKey)
        REFERENCES Snow_Promotion(PromotionKey)
);

# Star Schema: Supermarket Sales

One central fact table (`Star_FactSales`) connected directly to 5 dimension tables.

Legend: `PK` = Primary Key, `FK` = Foreign Key, `▲ ▼` = arrow points to the dimension table (the PK side).

```
┌────────────────────────┐   ┌────────────────────────┐   ┌────────────────────────┐
│ Star_DimTime           │   │ Star_DimProduct        │   │ Star_DimStore          │
├────────────────────────┤   ├────────────────────────┤   ├────────────────────────┤
│ PK TimeKey             │   │ PK ProductKey          │   │ PK StoreKey            │
│    FullDate            │   │    ProductName         │   │    StoreName           │
│    DayNumber           │   │    CategoryName        │   │    City                │
│    MonthName           │   │    BrandName           │   │    StateName           │
│    QuarterNumber       │   │                        │   │                        │
│    CalendarYear        │   │                        │   │                        │
└───────────┬────────────┘   └───────────┬────────────┘   └───────────┬────────────┘
            ▲                            ▲                            ▲
            │                            │                            │
            │ TimeKey                    │ ProductKey                 │ StoreKey
            │                            │                            │
      ┌─────┴────────────────────────────┴────────────────────────────┴──────┐
      │ Star_FactSales  (FACT TABLE)                                         │
      ├──────────────────────────────────────────────────────────────────────┤
      │ Keys                               Measures                          │
      │ PK SalesKey                          QuantitySold                    │
      │ FK TimeKey                           SalesAmount                     │
      │ FK ProductKey                        DiscountAmount                  │
      │ FK StoreKey                          Profit                          │
      │ FK CustomerKey                                                       │
      │ FK PromotionKey                                                      │
      └───────────────────┬──────────────────────────────┬───────────────────┘
                          │                              │
                          │ CustomerKey                  │ PromotionKey
                          ▼                              ▼
              ┌────────────────────────┐     ┌────────────────────────┐
              │ Star_DimCustomer       │     │ Star_DimPromotion      │
              ├────────────────────────┤     ├────────────────────────┤
              │ PK CustomerKey         │     │ PK PromotionKey        │
              │    CustomerName        │     │    PromotionName       │
              │    CustomerType        │     │    PromotionType       │
              │    Email               │     │    StartDate           │
              │                        │     │    EndDate             │
              └────────────────────────┘     └────────────────────────┘
```

## Connections

| Fact Table FK                  | Dimension Table PK                |
|--------------------------------|-----------------------------------|
| Star_FactSales.TimeKey         | Star_DimTime.TimeKey              |
| Star_FactSales.ProductKey      | Star_DimProduct.ProductKey        |
| Star_FactSales.StoreKey        | Star_DimStore.StoreKey            |
| Star_FactSales.CustomerKey     | Star_DimCustomer.CustomerKey      |
| Star_FactSales.PromotionKey    | Star_DimPromotion.PromotionKey    |

Total: 6 tables (1 fact + 5 dimensions). All dimensions are denormalized, so Category and Brand sit inside `Star_DimProduct`, and City and State sit inside `Star_DimStore`.













# Snowflake Schema: Supermarket Sales

Same fact table, but `Product` and `Store` are normalized into sub-dimensions:

- Product → Category and Product → Brand
- Store → City → State

`Snow_Time`, `Snow_Customer` and `Snow_Promotion` stay as single tables.

Legend: `PK` = Primary Key, `FK` = Foreign Key, `▲ ▼` = arrow points to the parent table (the PK side).

```
┌──────────────────────┐     ┌──────────────────────┐     ┌──────────────────────┐
│ Snow_Time            │     │ Snow_Customer        │     │ Snow_Promotion       │
├──────────────────────┤     ├──────────────────────┤     ├──────────────────────┤
│ PK TimeKey           │     │ PK CustomerKey       │     │ PK PromotionKey      │
│    FullDate          │     │    CustomerName      │     │    PromotionName     │
│    DayNumber         │     │    CustomerType      │     │    PromotionType     │
│    MonthName         │     │    Email             │     │    StartDate         │
│    QuarterNumber     │     └──────────┬───────────┘     │    EndDate           │
│    CalendarYear      │                ▲                 └──────────┬───────────┘
└──────────┬───────────┘                │                            ▲
           ▲                            │                            │
           │                            │                            │
           │                            │                            │
           │ TimeKey                    │ CustomerKey                │ PromotionKey
           │                            │                            │
           │                            │                            │
┌──────────┴────────────────────────────┴────────────────────────────┴───────────┐
│ Snow_FactSales  (FACT TABLE)                                                   │
├────────────────────────────────────────────────────────────────────────────────┤
│ Keys                                    Measures                               │
│ PK SalesKey                               QuantitySold                         │
│ FK TimeKey                                SalesAmount                          │
│ FK ProductKey                             DiscountAmount                       │
│ FK StoreKey                               Profit                               │
│ FK CustomerKey                                                                 │
│ FK PromotionKey                                                                │
└────────────────────────┬───────────────────────────────────────────┬───────────┘
                         │                                           │
                         │ ProductKey                                │ StoreKey
                         ▼                                           ▼
              ┌──────────────────────┐                    ┌──────────────────────┐
              │ Snow_Product         │                    │ Snow_Store           │
              ├──────────────────────┤                    ├──────────────────────┤
              │ PK ProductKey        │                    │ PK StoreKey          │
              │    ProductName       │                    │    StoreName         │
              │ FK CategoryKey       │                    │ FK CityKey           │
              │ FK BrandKey          │                    └──────────┬───────────┘
              └──────────┬───────────┘                               │ CityKey
                         │                                           ▼
           ┌─────────────┴──────────────┐                 ┌──────────────────────┐
           ▼ CategoryKey                ▼ BrandKey        │ Snow_City            │
┌──────────────────────┐     ┌──────────────────────┐     ├──────────────────────┤
│ Snow_Category        │     │ Snow_Brand           │     │ PK CityKey           │
├──────────────────────┤     ├──────────────────────┤     │    CityName          │
│ PK CategoryKey       │     │ PK BrandKey          │     │ FK StateKey          │
│    CategoryName      │     │    BrandName         │     └──────────┬───────────┘
└──────────────────────┘     └──────────────────────┘                │ StateKey
                                                                     ▼
                                                          ┌──────────────────────┐
                                                          │ Snow_State           │
                                                          ├──────────────────────┤
                                                          │ PK StateKey          │
                                                          │    StateName         │
                                                          └──────────────────────┘
```
