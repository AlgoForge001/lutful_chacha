PRAGMA foreign_keys = ON;

-- =========================================
-- PART 1: CREATE TABLES
-- =========================================

DROP TABLE IF EXISTS FactSales;
DROP TABLE IF EXISTS DimTime;
DROP TABLE IF EXISTS DimCustomer;
DROP TABLE IF EXISTS DimProduct;
DROP TABLE IF EXISTS DimLocation;

CREATE TABLE DimTime (
    TimeKey INTEGER PRIMARY KEY,
    Day INTEGER NOT NULL,
    Month INTEGER NOT NULL,
    Quarter INTEGER NOT NULL,
    Year INTEGER NOT NULL
);

CREATE TABLE DimCustomer (
    CustomerKey INTEGER PRIMARY KEY,
    CustomerName TEXT NOT NULL,
    Segment TEXT NOT NULL
);

CREATE TABLE DimProduct (
    ProductKey INTEGER PRIMARY KEY,
    ProductName TEXT NOT NULL,
    Category TEXT NOT NULL
);

CREATE TABLE DimLocation (
    LocationKey INTEGER PRIMARY KEY,
    City TEXT NOT NULL,
    State TEXT NOT NULL,
    Region TEXT NOT NULL
);

CREATE TABLE FactSales (
    SalesKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    CustomerKey INTEGER NOT NULL,
    ProductKey INTEGER NOT NULL,
    LocationKey INTEGER NOT NULL,
    QuantitySold INTEGER NOT NULL,
    SalesAmount REAL NOT NULL,
    FOREIGN KEY (TimeKey) REFERENCES DimTime(TimeKey),
    FOREIGN KEY (CustomerKey) REFERENCES DimCustomer(CustomerKey),
    FOREIGN KEY (ProductKey) REFERENCES DimProduct(ProductKey),
    FOREIGN KEY (LocationKey) REFERENCES DimLocation(LocationKey)
);

-- =========================================
-- PART 8: SLICE OPERATION
-- Customer segment = Retail
-- =========================================

SELECT
    C.Segment,
    C.CustomerName,
    SUM(F.QuantitySold) AS TotalQuantity,
    SUM(F.SalesAmount) AS TotalSales
FROM FactSales F
JOIN DimCustomer C ON F.CustomerKey = C.CustomerKey
WHERE C.Segment = 'Retail'
GROUP BY C.Segment, C.CustomerName
ORDER BY C.CustomerName;

-- Business question:
-- What are the total quantities and sales generated by Retail customers?

-- =========================================
-- PART 9: DICE OPERATION
-- Locations: Mumbai and Pune
-- Products: Electronics category
-- Measure condition: SalesAmount > 50000
-- =========================================

SELECT
    L.City,
    P.ProductName,
    P.Category,
    SUM(F.QuantitySold) AS TotalQuantity,
    SUM(F.SalesAmount) AS TotalSales
FROM FactSales F
JOIN DimLocation L ON F.LocationKey = L.LocationKey
JOIN DimProduct P ON F.ProductKey = P.ProductKey
WHERE L.City IN ('Mumbai', 'Pune')
  AND P.Category = 'Electronics'
GROUP BY L.City, P.ProductName, P.Category
HAVING SUM(F.SalesAmount) > 50000
ORDER BY L.City, P.ProductName;

-- Business question:
-- Which Electronics products generated more than 50000 in sales in Mumbai or Pune?

-- =========================================
-- PART 10: ROLL-UP OPERATION
-- Day -> Month -> Year
-- SQLite alternative to WITH ROLLUP
-- =========================================

-- Day-level totals
SELECT
    'Day' AS Level,
    T.Year,
    T.Month,
    T.Day,
    SUM(F.SalesAmount) AS TotalSales
FROM FactSales F
JOIN DimTime T ON F.TimeKey = T.TimeKey
GROUP BY T.Year, T.Month, T.Day

UNION ALL

-- Month-level totals
SELECT
    'Month' AS Level,
    T.Year,
    T.Month,
    NULL AS Day,
    SUM(F.SalesAmount) AS TotalSales
FROM FactSales F
JOIN DimTime T ON F.TimeKey = T.TimeKey
GROUP BY T.Year, T.Month

UNION ALL

-- Year-level totals
SELECT
    'Year' AS Level,
    T.Year,
    NULL AS Month,
    NULL AS Day,
    SUM(F.SalesAmount) AS TotalSales
FROM FactSales F
JOIN DimTime T ON F.TimeKey = T.TimeKey
GROUP BY T.Year
ORDER BY Year, Month, Day;

-- Business question:
-- How do sales totals aggregate from each day to each month and year?

-- =========================================
-- PART 11: DRILL-DOWN OPERATION
-- Year -> Month -> Day
-- =========================================

SELECT
    T.Year,
    T.Month,
    T.Day,
    SUM(F.QuantitySold) AS TotalQuantity,
    SUM(F.SalesAmount) AS TotalSales
FROM FactSales F
JOIN DimTime T ON F.TimeKey = T.TimeKey
GROUP BY T.Year, T.Month, T.Day
ORDER BY T.Year, T.Month, T.Day;

-- Business question:
-- Which months and days contribute to each year's sales?

-- =========================================
-- PART 12: PIVOT OPERATION
-- Customer segments become columns
-- =========================================

SELECT
    T.Year,
    SUM(CASE WHEN C.Segment = 'Retail'
             THEN F.SalesAmount ELSE 0 END) AS RetailSales,
    SUM(CASE WHEN C.Segment = 'Wholesale'
             THEN F.SalesAmount ELSE 0 END) AS WholesaleSales,
    SUM(CASE WHEN C.Segment = 'Corporate'
             THEN F.SalesAmount ELSE 0 END) AS CorporateSales
FROM FactSales F
JOIN DimTime T ON F.TimeKey = T.TimeKey
JOIN DimCustomer C ON F.CustomerKey = C.CustomerKey
GROUP BY T.Year
ORDER BY T.Year;
