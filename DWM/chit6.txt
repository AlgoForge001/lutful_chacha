/* =====================================================
   PART A: CREATE STAR SCHEMA
   ===================================================== */

DROP TABLE IF EXISTS Star_FactAdmissions;
DROP TABLE IF EXISTS Star_DimCategory;
DROP TABLE IF EXISTS Star_DimFaculty;
DROP TABLE IF EXISTS Star_DimCourse;
DROP TABLE IF EXISTS Star_DimStudent;
DROP TABLE IF EXISTS Star_DimTime;


/* 1. TIME DIMENSION */

CREATE TABLE Star_DimTime (
    TimeKey INTEGER PRIMARY KEY,
    AcademicYear VARCHAR(20) NOT NULL,
    StartYear INTEGER,
    EndYear INTEGER
);


/* 2. STUDENT DIMENSION */

CREATE TABLE Star_DimStudent (
    StudentKey INTEGER PRIMARY KEY,
    StudentName VARCHAR(100),
    Gender VARCHAR(20),
    City VARCHAR(50),
    StateName VARCHAR(50)
);


/* 3. COURSE DIMENSION */

CREATE TABLE Star_DimCourse (
    CourseKey INTEGER PRIMARY KEY,
    CourseName VARCHAR(100),
    BranchName VARCHAR(100),
    DepartmentName VARCHAR(100)
);


/* 4. FACULTY DIMENSION */

CREATE TABLE Star_DimFaculty (
    FacultyKey INTEGER PRIMARY KEY,
    FacultyName VARCHAR(100),
    Designation VARCHAR(50)
);


/* 5. CATEGORY DIMENSION */

CREATE TABLE Star_DimCategory (
    CategoryKey INTEGER PRIMARY KEY,
    CategoryName VARCHAR(50),
    CategoryDescription VARCHAR(100)
);


/* 6. FACT TABLE */

CREATE TABLE Star_FactAdmissions (
    AdmissionFactKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    StudentKey INTEGER NOT NULL,
    CourseKey INTEGER NOT NULL,
    FacultyKey INTEGER NOT NULL,
    CategoryKey INTEGER NOT NULL,

    ApplicationsReceived INTEGER NOT NULL,
    SeatsFilled INTEGER NOT NULL,
    FeesCollected DECIMAL(12,2) NOT NULL,
    AverageMarks DECIMAL(5,2) NOT NULL,
    PassPercentage DECIMAL(5,2) NOT NULL,

    FOREIGN KEY (TimeKey)
        REFERENCES Star_DimTime(TimeKey),

    FOREIGN KEY (StudentKey)
        REFERENCES Star_DimStudent(StudentKey),

    FOREIGN KEY (CourseKey)
        REFERENCES Star_DimCourse(CourseKey),

    FOREIGN KEY (FacultyKey)
        REFERENCES Star_DimFaculty(FacultyKey),

    FOREIGN KEY (CategoryKey)
        REFERENCES Star_DimCategory(CategoryKey)
);

/* =====================================================
   PART D: STAR SCHEMA JOIN QUERY
   FACT + ALL DIMENSIONS
   ===================================================== */

SELECT
    T.AcademicYear,
    T.StartYear,
    T.EndYear,

    S.StudentName,
    S.Gender,
    S.City,
    S.StateName,

    C.CourseName,
    C.BranchName,
    C.DepartmentName,

    FCL.FacultyName,
    FCL.Designation,

    CAT.CategoryName,
    CAT.CategoryDescription,

    F.ApplicationsReceived,
    F.SeatsFilled,
    F.FeesCollected,
    F.AverageMarks,
    F.PassPercentage

FROM Star_FactAdmissions F

JOIN Star_DimTime T
    ON F.TimeKey = T.TimeKey

JOIN Star_DimStudent S
    ON F.StudentKey = S.StudentKey

JOIN Star_DimCourse C
    ON F.CourseKey = C.CourseKey

JOIN Star_DimFaculty FCL
    ON F.FacultyKey = FCL.FacultyKey

JOIN Star_DimCategory CAT
    ON F.CategoryKey = CAT.CategoryKey;


/* =====================================================
   PART E: CREATE SNOWFLAKE SCHEMA
   ===================================================== */

DROP TABLE IF EXISTS Snow_FactAdmissions;
DROP TABLE IF EXISTS Snow_Faculty;
DROP TABLE IF EXISTS Snow_Course;
DROP TABLE IF EXISTS Snow_Department;
DROP TABLE IF EXISTS Snow_Student;
DROP TABLE IF EXISTS Snow_Category;
DROP TABLE IF EXISTS Snow_City;
DROP TABLE IF EXISTS Snow_State;
DROP TABLE IF EXISTS Snow_Time;


/* 1. TIME DIMENSION */

CREATE TABLE Snow_Time (
    TimeKey INTEGER PRIMARY KEY,
    AcademicYear VARCHAR(20) NOT NULL,
    StartYear INTEGER,
    EndYear INTEGER
);


/* 2. STATE DIMENSION */

CREATE TABLE Snow_State (
    StateKey INTEGER PRIMARY KEY,
    StateName VARCHAR(50) NOT NULL
);


/* 3. CITY DIMENSION */

CREATE TABLE Snow_City (
    CityKey INTEGER PRIMARY KEY,
    CityName VARCHAR(50) NOT NULL,
    StateKey INTEGER NOT NULL,

    FOREIGN KEY (StateKey)
        REFERENCES Snow_State(StateKey)
);


/* 4. CATEGORY DIMENSION */

CREATE TABLE Snow_Category (
    CategoryKey INTEGER PRIMARY KEY,
    CategoryName VARCHAR(50),
    CategoryDescription VARCHAR(100)
);


/* 5. STUDENT DIMENSION */

CREATE TABLE Snow_Student (
    StudentKey INTEGER PRIMARY KEY,
    StudentName VARCHAR(100),
    Gender VARCHAR(20),
    CityKey INTEGER NOT NULL,
    CategoryKey INTEGER NOT NULL,

    FOREIGN KEY (CityKey)
        REFERENCES Snow_City(CityKey),

    FOREIGN KEY (CategoryKey)
        REFERENCES Snow_Category(CategoryKey)
);


/* 6. DEPARTMENT DIMENSION */

CREATE TABLE Snow_Department (
    DepartmentKey INTEGER PRIMARY KEY,
    DepartmentName VARCHAR(100) NOT NULL
);


/* 7. COURSE DIMENSION */

CREATE TABLE Snow_Course (
    CourseKey INTEGER PRIMARY KEY,
    CourseName VARCHAR(100),
    BranchName VARCHAR(100),
    DepartmentKey INTEGER NOT NULL,

    FOREIGN KEY (DepartmentKey)
        REFERENCES Snow_Department(DepartmentKey)
);


/* 8. FACULTY DIMENSION */

CREATE TABLE Snow_Faculty (
    FacultyKey INTEGER PRIMARY KEY,
    FacultyName VARCHAR(100),
    Designation VARCHAR(50),
    DepartmentKey INTEGER NOT NULL,

    FOREIGN KEY (DepartmentKey)
        REFERENCES Snow_Department(DepartmentKey)
);


/* 9. SNOWFLAKE FACT TABLE */

CREATE TABLE Snow_FactAdmissions (
    AdmissionFactKey INTEGER PRIMARY KEY,
    TimeKey INTEGER NOT NULL,
    StudentKey INTEGER NOT NULL,
    CourseKey INTEGER NOT NULL,
    FacultyKey INTEGER NOT NULL,
    CategoryKey INTEGER NOT NULL,

    ApplicationsReceived INTEGER NOT NULL,
    SeatsFilled INTEGER NOT NULL,
    FeesCollected DECIMAL(12,2) NOT NULL,
    AverageMarks DECIMAL(5,2) NOT NULL,
    PassPercentage DECIMAL(5,2) NOT NULL,

    FOREIGN KEY (TimeKey)
        REFERENCES Snow_Time(TimeKey),

    FOREIGN KEY (StudentKey)
        REFERENCES Snow_Student(StudentKey),

    FOREIGN KEY (CourseKey)
        REFERENCES Snow_Course(CourseKey),

    FOREIGN KEY (FacultyKey)
        REFERENCES Snow_Faculty(FacultyKey),

    FOREIGN KEY (CategoryKey)
        REFERENCES Snow_Category(CategoryKey)
);

# Star Schema: College Admissions and Results

One central fact table (`Star_FactAdmissions`) connected directly to 5 dimension tables.

Legend: `PK` = Primary Key, `FK` = Foreign Key, `▲ ▼` = arrow points to the dimension table (the PK side).

```
┌────────────────────────┐   ┌────────────────────────┐   ┌────────────────────────┐
│ Star_DimTime           │   │ Star_DimStudent        │   │ Star_DimCourse         │
├────────────────────────┤   ├────────────────────────┤   ├────────────────────────┤
│ PK TimeKey             │   │ PK StudentKey          │   │ PK CourseKey           │
│    AcademicYear        │   │    StudentName         │   │    CourseName          │
│    StartYear           │   │    Gender              │   │    BranchName          │
│    EndYear             │   │    City                │   │    DepartmentName      │
│                        │   │    StateName           │   │                        │
│                        │   │                        │   │                        │
└───────────┬────────────┘   └───────────┬────────────┘   └───────────┬────────────┘
            ▲                            ▲                            ▲
            │                            │                            │
            │ TimeKey                    │ StudentKey                 │ CourseKey
            │                            │                            │
      ┌─────┴────────────────────────────┴────────────────────────────┴──────┐
      │ Star_FactAdmissions  (FACT TABLE)                                    │
      ├──────────────────────────────────────────────────────────────────────┤
      │ Keys                               Measures                          │
      │ PK AdmissionFactKey                  ApplicationsReceived            │
      │ FK TimeKey                           SeatsFilled                     │
      │ FK StudentKey                        FeesCollected                   │
      │ FK CourseKey                         AverageMarks                    │
      │ FK FacultyKey                        PassPercentage                  │
      │ FK CategoryKey                                                       │
      └───────────────────┬──────────────────────────────┬───────────────────┘
                          │                              │
                          │ FacultyKey                   │ CategoryKey
                          ▼                              ▼
              ┌────────────────────────┐     ┌────────────────────────┐
              │ Star_DimFaculty        │     │ Star_DimCategory       │
              ├────────────────────────┤     ├────────────────────────┤
              │ PK FacultyKey          │     │ PK CategoryKey         │
              │    FacultyName         │     │    CategoryName        │
              │    Designation         │     │    CategoryDescription │
              └────────────────────────┘     └────────────────────────┘
```

## Connections

| Fact Table FK                      | Dimension Table PK                |
|------------------------------------|-----------------------------------|
| Star_FactAdmissions.TimeKey        | Star_DimTime.TimeKey              |
| Star_FactAdmissions.StudentKey     | Star_DimStudent.StudentKey        |
| Star_FactAdmissions.CourseKey      | Star_DimCourse.CourseKey          |
| Star_FactAdmissions.FacultyKey     | Star_DimFaculty.FacultyKey        |
| Star_FactAdmissions.CategoryKey    | Star_DimCategory.CategoryKey      |

Total: 6 tables (1 fact + 5 dimensions). All dimensions are denormalized: City and StateName sit inside `Star_DimStudent`, and BranchName and DepartmentName sit inside `Star_DimCourse`.















# Snowflake Schema: College Admissions and Results

Same fact table, but the dimensions are normalized into sub-dimensions:

- Student → City → State
- Student → Category (`Snow_Category` is also linked directly to the fact table)
- Course → Department
- Faculty → Department (`Snow_Department` is shared by `Snow_Course` and `Snow_Faculty`, and has no direct link to the fact table)

`Snow_Time` stays as a single table.

Legend: `PK` = Primary Key, `FK` = Foreign Key, `▲ ▼ ► ◄` = arrow points to the parent table (the PK side).

